rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    // Admin is either a custom claim or the admin email (verified accounts only
    // matter if sign-up is open; accounts are created by hand in the console).
    function isAdmin() {
      return request.auth != null &&
        (request.auth.token.get('admin', false) == true ||
         request.auth.token.get('email', '') == 'rahilrizvi0786110@gmail.com');
    }

    // Users can read/write only their own progress document; admin can read all.
    match /progress/{userId} {
      allow read, write: if request.auth != null && request.auth.uid == userId;
      allow read: if isAdmin();
    }

    // Grammar exercises are public (read-only) for logged-in users.
    match /grammar_exercises/{exerciseId} {
      allow read: if request.auth != null;
      allow write: if false; // only server-side scripts can create exercises
    }

    // What the grammar agent remembers about her. Written only by the daily job.
    match /grammar_memory/{docId} {
      allow read: if isAdmin();
      allow write: if false;
    }

    // Grammar attempt history: users can create/read their own; admin can read all.
    match /grammar_attempts/{attemptId} {
      allow read: if request.auth != null && resource.data.userId == request.auth.uid;
      allow create: if request.auth != null && request.resource.data.userId == request.auth.uid;
      allow update, delete: if false;
      allow read: if isAdmin();
    }
  }
}
